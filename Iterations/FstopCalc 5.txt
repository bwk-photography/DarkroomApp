<script>
let lastData = [];


window.addEventListener("load", () => {
    const viewModeSelect = document.getElementById("viewMode");

    const savedMode = localStorage.getItem("viewMode");
    const isDesktop = window.matchMedia("(min-width: 768px)").matches;

    let mode = savedMode ? savedMode : (isDesktop ? "table" : "cards");

    viewModeSelect.value = mode;

    // run initial render
    calculate();

    // save + recalc when mode changes
    viewModeSelect.addEventListener("change", () => {
        localStorage.setItem("viewMode", viewModeSelect.value);
        calculate();
    });

    // NEW: auto-recalc on all inputs
    setupAutoCalculate();
});


function setupAutoCalculate() {
    const inputs = [
        "baseTime",
        "increment",
        "steps",
        "viewMode"
    ];

    const trigger = () => calculate();

    inputs.forEach(id => {
        const el = document.getElementById(id);

        if (!el) return;

        el.addEventListener("change", trigger);

        // smoother UX for typing
        el.addEventListener("input", () => {
            clearTimeout(window._calcTimer);
            window._calcTimer = setTimeout(trigger, 150);
        });
    });
}


function calculate() {
	const baseTimeInput = document.getElementById('baseTime');

	let baseTime = parseFloat(baseTimeInput.value);

	// force whole seconds (handles pasted decimals too)
	baseTime = Math.round(baseTime);

	// reflect correction back to UI
	baseTimeInput.value = baseTime;

	//const baseTime = parseFloat(document.getElementById('baseTime').value);
	const increment = parseFloat(document.getElementById('increment').value);
	const steps = parseInt(document.getElementById('steps').value);
	const viewMode = document.getElementById("viewMode").value;

	const panel = document.getElementById("panel");
	const tbody = document.querySelector("#resultsTable tbody");

	panel.innerHTML = "";
	tbody.innerHTML = "";
	lastData = [];

	const half = Math.floor(steps / 2);
	let stripTime = 0;
	let stepCount = 0;

	for (let i = half; i >= -half; i--) {
		stepCount++;

	const stops = i * increment;
	const time = baseTime * Math.pow(2, stops);
	const isBase = Math.abs(time - baseTime) < 0.0001;

	const stops_next = (i - 1) * increment;
	const time_next = baseTime * Math.pow(2, stops_next);

	// on the final step, show the total time of the last exposure
	// for the rest, use the current exposure time minus the exposure time for the next step
	if (stepCount === steps) {
		stripTime = time;
	} else {
	        stripTime = time-time_next;
	}

	const rowData = {
		step: stepCount,
		stops,
		time,
		strip: stripTime,
		isBase
	};

	lastData.push(rowData);

	// ================= CARD VIEW =================
	if (viewMode === "cards") {
		const card = document.createElement("div");
		card.className = "card";
		if (isBase) card.classList.add("base");

		card.innerHTML = `
			<div><b>Step ${stepCount}</b></div>
			<div>Stops: ${isBase ? "BASE" : stops.toFixed(2)}</div>
			<div>Strip: ${stripTime.toFixed(1)}s</div>
			<div>Exposure: ${time.toFixed(1)}s</div>
		`;

		card.addEventListener("click", () => {
		document.querySelectorAll(".card").forEach(c => c.classList.remove("selected"));
		card.classList.add("selected");
	});

		panel.appendChild(card);
	}

	// ================= TABLE VIEW =================
	if (viewMode === "table") {
		const row = document.createElement("tr");

		if (isBase) row.classList.add("base-row");

		row.innerHTML = `
			<td>${stepCount}</td>
			<td>${isBase ? "BASE" : stops.toFixed(2)}</td>
			<td>${stripTime.toFixed(1)}</td>
			<td>${time.toFixed(1)}</td>
		`;

		row.addEventListener("click", () => {
		document.querySelectorAll("tr").forEach(r => r.classList.remove("selected-row"));
		row.classList.add("selected-row");
	});

		tbody.appendChild(row);
	}
}

	document.getElementById("panel").style.display =
        viewMode === "cards" ? "grid" : "none";

	document.getElementById("tableContainer").style.display =
	viewMode === "table" ? "block" : "none";
}


/* =========================
   PDF EXPORT (SIMPLE)
   Uses browser print-to-PDF
========================= */
function exportPDF() {
    const mode = document.getElementById("viewMode").value;

    const printWindow = window.open('', '', 'width=800,height=600');

    let html = `
    <html>
    <head>
    <title>F-Stop Export</title>
    <style>
        body { font-family: Arial; padding: 20px; }
        .card { border: 1px solid #000; padding: 10px; margin: 10px 0; }
        table { width: 100%; border-collapse: collapse; }
        td, th { border: 1px solid #000; padding: 8px; text-align: center; }
        .base { font-weight: bold; }
    </style>
    </head>
    <body>
    <h2>F-Stop Exposure Sheet</h2>
    `;

    if (mode === "cards") {
        lastData.forEach(d => {
            html += `
            <div class="card ${d.isBase ? 'base' : ''}">
                Step ${d.step}<br>
                Stops: ${d.isBase ? "BASE" : d.stops.toFixed(2)}<br>
                Exposure: ${d.time.toFixed(1)}s<br>
                Strip: ${d.strip.toFixed(1)}s
            </div>`;
        });
    } else {
        html += `<table><tr><th>Step</th><th>Stops</th><th>Exposure</th><th>Strip</th></tr>`;
        lastData.forEach(d => {
            html += `<tr>
                <td>${d.step}</td>
                <td>${d.isBase ? "BASE" : d.stops.toFixed(2)}</td>
                <td>${d.time.toFixed(1)}</td>
                <td>${d.strip.toFixed(1)}</td>
            </tr>`;
        });
        html += `</table>`;
    }

    html += `
    <script>
        window.onload = function() { window.print(); }
    <\/script>
    </body></html>`;

    printWindow.document.write(html);
    printWindow.document.close();
}
</script>
